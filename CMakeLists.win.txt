cmake_minimum_required(VERSION 3.16)
project(haxball_client C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/SDL2-2.30.9/x86_64-w64-mingw32/include
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/SDL2-2.30.9/x86_64-w64-mingw32/include/SDL2
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/libdatachannel-src/include
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/mingw64/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/stb
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

set(SOURCES
    src/main.cpp
    src/app.cpp
    src/ui/room_browser.cpp
    src/render/font.cpp
    src/render/renderer.cpp
    src/audio/sound.cpp
    src/physics/stadium.cpp
    src/physics/world.cpp
    src/assist/tactical_assistant.cpp
    src/net/room_list.cpp
    src/net/haxball_client.cpp
)

add_executable(haxball_client.exe ${SOURCES})

# Windows definitions
target_compile_definitions(haxball_client.exe PRIVATE
    WIN32_LEAN_AND_MEAN
    NOMINMAX
    _WIN32_WINNT=0x0601
)

target_link_directories(haxball_client.exe PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/SDL2-2.30.9/x86_64-w64-mingw32/lib
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/mingw64/lib
)

target_link_libraries(haxball_client.exe PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/libdatachannel-build/libdatachannel-static.a
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/libdatachannel-build/deps/usrsctp/usrsctplib/libusrsctp.a
    ${CMAKE_CURRENT_SOURCE_DIR}/win_deps/libdatachannel-build/deps/libjuice/libjuice-static.a
    mingw32
    SDL2main
    SDL2
    ssl
    crypto
    ws2_32
    crypt32
    iphlpapi
    bcrypt
    winmm
    imm32
    version
    setupapi
    m
)

set_target_properties(haxball_client.exe PROPERTIES
    LINK_FLAGS "-static-libgcc -static-libstdc++ -Wl,--subsystem,windows -municode"
)
